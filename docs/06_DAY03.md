### 1. `ugv_mon/parser/icd_parser.py` - ICD 파서

**역할**: ICD v1.0 규격에 따라 UDP 패킷 페이로드를 파싱하여 운용 상태 추출

**구현 내용**:

```python
"""
ICD v1.0 메시지 파서

역할:
- UDP 페이로드에서 ICD 메시지 추출
- 헤더 파싱 (TimeStamp, MSG ID, DataLength)
- 페이로드 파싱 (상태보고, Code 0x01)
- 체크섬 검증
- 시퀀스 번호 추출

ICD v1.0 메시지 구조:
- Header: 12 bytes (TimeStamp 4B, MSG ID 4B, Reserved 2B, DataLength 2B)
- Data: Variable (상태보고는 87 bytes)
- Checksum: 2 bytes (sum of all bytes except checksum & 0xFFFF)
"""

import struct
from dataclasses import dataclass
from typing import Optional
from enum import IntEnum

from ..config import config


class MsgCode(IntEnum):
    """ICD 메시지 코드"""
    STATUS_REPORT = 0x01
    # 추가 메시지 코드는 필요시 확장


@dataclass
class ICDHeader:
    """
    ICD 메시지 헤더 (12 bytes)
    
    Attributes:
        timestamp: 4-byte timestamp (Little Endian)
        sequence: 시퀀스 번호 (timestamp의 마지막 1 byte)
        source_id: Source ID (1 byte)
        dest_id: Destination ID (1 byte)
        msg_code: Message code (1 byte, 0x01 = 상태보고)
        ack_flag: Acknowledge flag (1 byte)
        reserved: Reserved field (2 bytes)
        data_length: Payload length (2 bytes, Little Endian)
    """
    timestamp: int
    sequence: int
    source_id: int
    dest_id: int
    msg_code: int
    ack_flag: int
    reserved: int
    data_length: int


@dataclass
class StatusReport:
    """
    상태보고 페이로드 (Code 0x01)
    
    Attributes:
        device_presence: 10개 장치 연결 비트 (bits 9-0)
        operation_mode: 운용모드 (bits 7-5 of VIC state byte)
        authority: 운용권한 (bits 3-2 of VIC state byte)
        driving_state: 주행상태 (bits 1-0 of VIC state byte)
        emergency_bits: 비상정지 원인 (bits 15-6 of emergency word)
    """
    device_presence: int
    operation_mode: int
    authority: int
    driving_state: int
    emergency_bits: int


@dataclass
class ParseResult:
    """
    파싱 결과
    
    Attributes:
        success: 파싱 성공 여부
        header: 파싱된 헤더 (None if failed)
        payload: 파싱된 페이로드 (None if failed or not status report)
        checksum_ok: 체크섬 검증 결과
        error: 에러 메시지 (if failed)
    """
    success: bool
    header: Optional[ICDHeader] = None
    payload: Optional[StatusReport] = None
    checksum_ok: bool = False
    error: str = ""


class ICDParser:
    """
    ICD v1.0 메시지 파서
    
    사용법:
        parser = ICDParser()
        result = parser.parse(udp_payload_bytes)
        if result.success:
            print(f"Sequence: {result.header.sequence}")
            print(f"Mode: {result.payload.operation_mode}")
    """
    
    HEADER_SIZE = 12
    CHECKSUM_SIZE = 2
    
    def parse(self, data: bytes) -> ParseResult:
        """
        전체 ICD 메시지 파싱
        
        Args:
            data: UDP 페이로드 (ICD 메시지 전체, 체크섬 포함)
            
        Returns:
            ParseResult with header, payload, status
        """
        # 1. 최소 길이 검증
        if len(data) < self.HEADER_SIZE + self.CHECKSUM_SIZE:
            return ParseResult(
                success=False,
                error=f"Packet too short: {len(data)} bytes (minimum {self.HEADER_SIZE + self.CHECKSUM_SIZE})"
            )
        
        # 2. 체크섬 검증
        checksum_ok = self._verify_checksum(data)
        
        # 3. 헤더 파싱
        header = self._parse_header(data[:self.HEADER_SIZE])
        if header is None:
            return ParseResult(
                success=False,
                checksum_ok=checksum_ok,
                error="Header parse failed"
            )
        
        # 4. 데이터 길이 검증
        expected_len = self.HEADER_SIZE + header.data_length + self.CHECKSUM_SIZE
        if len(data) != expected_len:
            return ParseResult(
                success=False,
                header=header,
                checksum_ok=checksum_ok,
                error=f"Length mismatch: expected {expected_len}, got {len(data)}"
            )
        
        # 5. 페이로드 파싱 (상태보고만 현재 지원)
        payload = None
        if header.msg_code == MsgCode.STATUS_REPORT:
            payload_data = data[self.HEADER_SIZE:-self.CHECKSUM_SIZE]
            payload = self._parse_status_report(payload_data)
            if payload is None:
                return ParseResult(
                    success=False,
                    header=header,
                    checksum_ok=checksum_ok,
                    error="Status report payload parse failed"
                )
        
        return ParseResult(
            success=True,
            header=header,
            payload=payload,
            checksum_ok=checksum_ok,
        )
    
    def _parse_header(self, data: bytes) -> Optional[ICDHeader]:
        """
        헤더 파싱 (12 bytes, Little Endian)
        
        Args:
            data: 헤더 데이터 (12 bytes)
            
        Returns:
            ICDHeader 객체 또는 None (파싱 실패 시)
        """
        if len(data) < self.HEADER_SIZE:
            return None
        
        try:
            # TimeStamp (4 bytes, Little Endian)
            timestamp = struct.unpack('<I', data[0:4])[0]
            sequence = data[3]  # 마지막 1 byte가 시퀀스 번호
            
            # MSG ID (4 bytes)
            source_id = data[4]
            dest_id = data[5]
            msg_code = data[6]
            ack_flag = data[7]
            
            # Reserved (2 bytes, Little Endian)
            reserved = struct.unpack('<H', data[8:10])[0]
            
            # DataLength (2 bytes, Little Endian)
            data_length = struct.unpack('<H', data[10:12])[0]
            
            return ICDHeader(
                timestamp=timestamp,
                sequence=sequence,
                source_id=source_id,
                dest_id=dest_id,
                msg_code=msg_code,
                ack_flag=ack_flag,
                reserved=reserved,
                data_length=data_length,
            )
        except Exception as e:
            return None
    
    def _parse_status_report(self, data: bytes) -> Optional[StatusReport]:
        """
        상태보고 페이로드 파싱 (Code 0x01)
        
        페이로드 구조 (87 bytes):
        - Offset 0-1: 구성품 연결 여부 (uint16, bits 9-0)
        - Offset 2: VIC 운용 상태 (uint8, bits 7-5=모드, 3-2=권한, 1-0=주행)
        - Offset 3-4: 비상정지 원인 (uint16, bits 15-6)
        - 나머지: 추가 필드 (현재 미사용)
        
        Args:
            data: 페이로드 데이터 (DataLength bytes)
            
        Returns:
            StatusReport 객체 또는 None (파싱 실패 시)
        """
        if len(data) < 5:  # 최소 5 bytes 필요
            return None
        
        try:
            # 구성품 연결 여부 (offset 0-1, uint16, Little Endian)
            device_presence = struct.unpack('<H', data[0:2])[0]
            
            # VIC 운용 상태 (offset 2, uint8)
            vic_state = data[2]
            operation_mode = (vic_state >> 5) & 0x07  # bits 7-5
            authority = (vic_state >> 2) & 0x03       # bits 3-2
            driving_state = vic_state & 0x03          # bits 1-0
            
            # 비상정지 원인 (offset 3-4, uint16, bits 15-6)
            emergency_word = struct.unpack('<H', data[3:5])[0]
            emergency_bits = (emergency_word >> 6) & 0x3FF  # bits 15-6
            
            return StatusReport(
                device_presence=device_presence,
                operation_mode=operation_mode,
                authority=authority,
                driving_state=driving_state,
                emergency_bits=emergency_bits,
            )
        except Exception as e:
            return None
    
    def _verify_checksum(self, data: bytes) -> bool:
        """
        체크섬 검증
        
        체크섬 규칙:
        - 체크섬 = (체크섬 필드를 제외한 모든 바이트 합) & 0xFFFF
        
        Args:
            data: 전체 ICD 메시지 (체크섬 포함)
            
        Returns:
            True if checksum matches, False otherwise
        """
        if len(data) < self.CHECKSUM_SIZE:
            return False
        
        try:
            # 체크섬 필드 제외
            data_without_checksum = data[:-self.CHECKSUM_SIZE]
            
            # 수신된 체크섬 추출 (2 bytes, Little Endian)
            received_checksum = struct.unpack('<H', data[-self.CHECKSUM_SIZE:])[0]
            
            # 계산된 체크섬
            calculated_checksum = sum(data_without_checksum) & 0xFFFF
            
            return calculated_checksum == received_checksum
        except Exception:
            return False
```

**설명 및 근거**:
- **Little Endian**: ICD 규격에 따라 모든 다중 바이트 필드는 Little Endian 사용
- **시퀀스 번호**: timestamp의 마지막 1 byte 사용 (멘토 확정)
- **체크섬 검증**: 패킷 무결성 보장, 파싱 전에 검증
- **에러 처리**: 파싱 실패 시 명확한 에러 메시지 제공
- **확장성**: 다른 메시지 코드(Code 0x01 외) 지원 가능하도록 구조화

from ugv_mon.parser.icd_parser import ICDParser

sniffer = PacketSniffer('lo')

for i in range(10):
    packet = sniffer.get_packet(timeout=1.0)
    if packet:
        # UDP 페이로드 추출

            continue

        if result.header:

        if result.payload:
            print(f'  Mode: {result.payload.operation_mode}')

            print(f'  Driving: {result.payload.driving_state}')
            print(f'  Devices: {bin(result.payload.device_presence)}')

## 🔧 문제 해결

### 문제: 파싱 실패 (success=False)

**가능한 원인**:
1. 패킷 길이 불일치
2. 체크섬 불일치
3. 헤더 구조 불일치

**해결**:
- Wireshark에서 패킷 hex dump 확인
- `data_length` 값과 실제 페이로드 길이 비교
- 체크섬 계산 로직 검증

### 문제: 비트 추출 오류

**원인**: 비트 마스킹 로직 오류

**해결**:
```python
# 올바른 비트 추출 예시
operation_mode = (vic_state >> 5) & 0x07  # bits 7-5
authority = (vic_state >> 2) & 0x03       # bits 3-2
driving_state = vic_state & 0x03          # bits 1-0
```
