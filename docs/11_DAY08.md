```python
"""
Chart Components Module.

Plotly 기반 차트 컴포넌트를 정의합니다.

차트 종류:
1. 통신 품질 차트 (PPS + 지터 스택)
2. 가용성 타임라인 (Up/Down 세그먼트)

특징:
- 시간 범위 선택 (30초/1분/5분)
- 상호작용 비활성화 (모니터링 전용)
- 최적화된 렌더링
"""

from typing import Dict, List
import plotly.graph_objects as go
from plotly.subplots import make_subplots


def get_chart_colors() -> Dict[str, str]:
    """
    차트 색상 팔레트.
    
    Returns:
        색상 딕셔너리
    """
    return {
        "primary": "#1e40af",    # PPS (파란색)
        "secondary": "#7c3aed",  # 지터 (보라색)
        "success": "#10b981",    # Up (초록색)
        "danger": "#ef4444",     # Down (빨간색)
        "grid": "#e5e7eb",       # 그리드 (밝은 회색)
        "text": "#374151",       # 텍스트 (어두운 회색)
        "bg": "#ffffff",         # 배경 (흰색)
    }


def create_communication_chart(
    data: List[Dict],
    p95: float,
    p99: float,
    time_range_seconds: int = 60
) -> go.Figure:
    """
    통신 품질 차트 생성 (PPS + 지터).
    
    두 개의 서브플롯으로 구성:
    - 위: PPS (Packets Per Second)
    - 아래: Jitter (밀리초)
    
    Args:
        data: 차트 데이터 포인트 리스트
              [{"timestamp": "HH:MM:SS", "pps": int, "jitter": float}, ...]
        p95: 지터 P95 참조선 값
        p99: 지터 P99 참조선 값
        time_range_seconds: 표시할 시간 범위 (초)
        
    Returns:
        Plotly Figure 객체
    """
    colors = get_chart_colors()
    
    # 시간 범위에 따른 데이터 필터링
    # 2초 폴링 기준: 30초=15개, 60초=30개, 300초=150개
    points_per_second = 0.5  # 2초당 1개
    max_points = int(time_range_seconds * points_per_second)
    filtered_data = data[-max_points:] if len(data) > max_points else data
    
    # 데이터 추출
    timestamps = [d.get("timestamp", "") for d in filtered_data]
    pps_values = [d.get("pps", 0) for d in filtered_data]
    jitter_values = [d.get("jitter", 0) for d in filtered_data]
    
    # 서브플롯 생성
    fig = make_subplots(
        rows=2,
        cols=1,
        subplot_titles=("패킷 수신율 (PPS)", "지터 (Jitter, ms)"),
        vertical_spacing=0.15,  # 서브플롯 간격
        row_heights=[0.5, 0.5],
    )
    
    # PPS 트레이스 (위쪽)
    fig.add_trace(
        go.Scatter(
            x=timestamps,
            y=pps_values,
            mode="lines",
            name="PPS",
            line=dict(color=colors["primary"], width=2),
            fill="tozeroy",
            fillcolor="rgba(30, 64, 175, 0.1)",
            hovertemplate="시간: %{x}<br>PPS: %{y:,}<extra></extra>",
        ),
        row=1,
        col=1,
    )
    
    # Jitter 트레이스 (아래쪽)
    fig.add_trace(
        go.Scatter(
            x=timestamps,
            y=jitter_values,
            mode="lines",
            name="Jitter",
            line=dict(color=colors["secondary"], width=2),
            fill="tozeroy",
            fillcolor="rgba(124, 58, 237, 0.1)",
            hovertemplate="시간: %{x}<br>지터: %{y:.1f}ms<extra></extra>",
        ),
        row=2,
        col=1,
    )
    
    # P95/P99 참조선 (지터 차트에만)
    if p95 > 0:
        fig.add_hline(
            y=p95,
            line_dash="dash",
            line_color="orange",
            annotation_text=f"P95: {p95:.1f}ms",
            annotation_position="right",
            row=2,
            col=1,
        )
    
    if p99 > 0:
        fig.add_hline(
            y=p99,
            line_dash="dot",
            line_color="red",
            annotation_text=f"P99: {p99:.1f}ms",
            annotation_position="right",
            row=2,
            col=1,
        )
    
    # 시간 범위에 따른 X축 틱 수 조정
    nticks_map = {30: 4, 60: 5, 300: 6}
    current_nticks = nticks_map.get(time_range_seconds, 5)
    
    # 레이아웃 설정
    fig.update_layout(
        height=320,
        margin=dict(l=50, r=20, t=40, b=30),
        showlegend=False,  # 범례는 외부에서 표시
        plot_bgcolor=colors["bg"],
        paper_bgcolor=colors["bg"],
        font=dict(family="sans-serif", size=11, color=colors["text"]),
        hovermode="x unified",
        dragmode=False,  # 드래그 비활성화
    )
    
    # Y축: 자동 스케일링
    fig.update_yaxes(
        rangemode="normal",
        gridcolor=colors["grid"],
        gridwidth=1,
    )
    
    # X축: 시간 포맷
    fig.update_xaxes(
        showgrid=True,
        gridcolor=colors["grid"],
        gridwidth=1,
        nticks=current_nticks,
        tickformat="%H:%M:%S",
        tickangle=0,
    )
    
    return fig


def create_availability_timeline(segments: List[Dict]) -> go.Figure:
    """
    가용성 타임라인 생성.
    
    Up(초록)/Down(빨강) 세그먼트를 수평 바로 표시합니다.
    
    Args:
        segments: 세그먼트 리스트
                  [{"start": float, "end": float, "status": "up"|"down"}, ...]
                  
    Returns:
        Plotly Figure 객체
    """
    colors = get_chart_colors()
    
    fig = go.Figure()
    
    if not segments:
        # 빈 차트
        fig.add_trace(go.Scatter(x=[], y=[], mode="lines"))
    else:
        for seg in segments:
            start = seg.get("start", 0)
            end = seg.get("end", 0)
            status = seg.get("status", "up")
            
            # 유효성 검사
            if end <= start or end <= 0:
                continue
            
            # 색상 선택
            color = colors["success"] if status == "up" else colors["danger"]
            status_text = "정상" if status == "up" else "중단"
            duration = end - start
            
            # 세그먼트 전체에서 hover 가능하도록 중간점들 생성
            x_points = [start]
            y_points = [0]
            
            # 50초 간격으로 중간점 추가
            current = start + 50
            while current < end:
                x_points.append(current)
                y_points.append(0)
                current += 50
            
            x_points.append(end)
            y_points.append(0)
            
            # Scatter로 두꺼운 선 그리기
            fig.add_trace(
                go.Scatter(
                    x=x_points,
                    y=y_points,
                    mode="lines",
                    line=dict(color=color, width=40),  # 두꺼운 선
                    fill="tozeroy",
                    fillcolor=color,
                    name=status_text,
                    hovertemplate=(
                        f"<b>{status_text}</b><br>"
                        f"시작: {start:.0f}초<br>"
                        f"종료: {end:.0f}초<br>"
                        f"지속: {duration:.0f}초"
                        "<extra></extra>"
                    ),
                    showlegend=False,
                )
            )
    
    # 레이아웃
    fig.update_layout(
        height=100,
        margin=dict(l=50, r=20, t=10, b=50),
        plot_bgcolor=colors["bg"],
        paper_bgcolor=colors["bg"],
        xaxis=dict(
            title=dict(text="Time (seconds)", standoff=10),
            showgrid=True,
            gridcolor=colors["grid"],
            range=[0, 3600],
        ),
        yaxis=dict(
            range=[-0.3, 0.3],
            showticklabels=False,
            showgrid=False,
        ),
        showlegend=False,
        hovermode="closest",
        dragmode=False,
    )
    
    return fig
```

## 🧪 테스트 방법

### 1. 시간 범위 선택 테스트

```
1. 대시보드 실행
2. 차트 상단의 "30초", "1분", "5분" 버튼 클릭
3. 차트 데이터 범위가 변경되는지 확인
4. X축 레이블 수가 적절하게 조정되는지 확인
```

### 2. 실시간 값 표시 테스트

```
1. 대시보드 실행
2. 차트 하단의 "PPS:", "지터:" 값 확인
3. 2초마다 값이 갱신되는지 확인
```

### 3. 가용성 타임라인 테스트

```
1. 초록색(Up), 빨간색(Down) 세그먼트 표시 확인
2. 세그먼트 위에 마우스 hover 시 정보 표시 확인
3. 드래그/클릭 시 차트 변경되지 않는지 확인
```
