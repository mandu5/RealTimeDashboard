## 💻 코드 구현 내용

### `ugv_mon/layouts/charts.py` 수정

**주요 수정 사항**:

1. **차트 데이터 소스 변경**: Mock 데이터 → LiveDataProvider
2. **Range selector 추가**: 시간 범위 선택 (30s/1m/5m)
3. **Hover 정보 강화**: 상세 정보 표시
4. **참조선 추가**: 지터 P95/P99 라인

**수정 예시**:

```python
def create_communication_chart(
    data: List[Dict],
    p95: float,
    p99: float
) -> go.Figure:
    """
    통신 품질 차트 생성 (PPS + 지터)
    
    수정 사항:
    - Range selector 추가
    - Hover 정보 강화
    - P95/P99 참조선 명확화
    """
    colors = get_chart_colors()
    
    fig = make_subplots(
        rows=2, cols=1,
        subplot_titles=("패킷 수신율 (PPS)", "지터 (Jitter, ms)"),
        vertical_spacing=0.12,
        row_heights=[0.5, 0.5],
    )
    
    # 데이터 추출
    timestamps = [d.get("timestamp", "") for d in data]
    pps_values = [d.get("pps", 0) for d in data]
    jitter_values = [d.get("jitter", 0) for d in data]
    
    # PPS 차트
    fig.add_trace(
        go.Scatter(
            x=timestamps,
            y=pps_values,
            mode="lines",
            name="PPS",
            line=dict(color=colors["primary"], width=2),
            fill="tozeroy",
            fillcolor=colors["fill_primary"],
            hovertemplate="<b>PPS:</b> %{y}<br><b>Time:</b> %{x}<extra></extra>",
        ),
        row=1, col=1,
    )
    
    # 지터 차트
    fig.add_trace(
        go.Scatter(
            x=timestamps,
            y=jitter_values,
            mode="lines",
            name="Jitter",
            line=dict(color=colors["secondary"], width=2),
            fill="tozeroy",
            fillcolor=colors["fill_secondary"],
            hovertemplate="<b>Jitter:</b> %{y}ms<br><b>Time:</b> %{x}<extra></extra>",
        ),
        row=2, col=1,
    )
    
    # P95/P99 참조선
    fig.add_hline(
        y=p95,
        line_dash="dash",
        line_color=colors["p95_line"],
        annotation_text=f"P95: {p95}ms",
        annotation_position="right",
        row=2, col=1,
    )
    fig.add_hline(
        y=p99,
        line_dash="dash",
        line_color=colors["p99_line"],
        annotation_text=f"P99: {p99}ms",
        annotation_position="right",
        row=2, col=1,
    )
    
    # 레이아웃 설정
    fig.update_layout(
        height=400,
        margin=dict(l=40, r=40, t=40, b=40),
        plot_bgcolor="white",
        paper_bgcolor="white",
        font=dict(family="Inter, sans-serif", size=11),
        showlegend=False,
        hovermode="x unified",
        # Range selector 추가
        xaxis=dict(
            rangeselector=dict(
                buttons=list([
                    dict(count=30, label="30s", step="second", stepmode="backward"),
                    dict(count=60, label="1m", step="second", stepmode="backward"),
                    dict(count=300, label="5m", step="second", stepmode="backward"),
                ])
            ),
        ),
    )
    
    fig.update_xaxes(showgrid=True, gridwidth=1, gridcolor=colors["grid"])
    fig.update_yaxes(showgrid=True, gridwidth=1, gridcolor=colors["grid"])
    
    return fig
```
