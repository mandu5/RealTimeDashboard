### 2. `ugv_mon/analysis/stats_calculator.py`

```python
"""
Statistics Calculator Module.

실시간 통계 계산 기능을 제공합니다:
- PPS (Packets Per Second)
- 지터 (P95, P99 백분위수)
- 패킷 손실률
- 가용성 (5분/1시간)

슬라이딩 윈도우 방식으로 최근 데이터만 사용합니다.
"""

from collections import deque
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import List, Optional, Tuple
import threading


@dataclass
class PacketRecord:
    """
    패킷 수신 기록.
    
    지터 및 PPS 계산에 사용됩니다.
    """
    timestamp: datetime      # 수신 시간
    sequence: int           # 시퀀스 번호
    size: int               # 패킷 크기


class StatsCalculator:
    """
    통계 계산기.
    
    슬라이딩 윈도우 방식으로 실시간 통계를 계산합니다.
    스레드 안전합니다.
    
    Usage:
        calc = StatsCalculator(window_sec=60)
        calc.record_packet(datetime.now(), seq=1, size=100)
        
        pps = calc.get_pps()
        p95, p99 = calc.get_jitter_percentiles()
    """
    
    def __init__(
        self,
        window_sec: int = 60,
        jitter_buffer_size: int = 1000
    ):
        """
        StatsCalculator 초기화.
        
        Args:
            window_sec: 통계 계산 윈도우 (초)
            jitter_buffer_size: 지터 계산용 버퍼 크기
        """
        self._window_sec = window_sec
        self._lock = threading.Lock()
        
        # 패킷 기록 (시간순)
        self._packets: deque = deque()
        
        # 지터 값 버퍼 (최근 N개)
        self._jitter_buffer: deque = deque(maxlen=jitter_buffer_size)
        
        # 이전 패킷 정보 (지터 계산용)
        self._prev_packet_time: Optional[datetime] = None
        self._prev_sequence: Optional[int] = None
        
        # 누적 통계
        self._total_packets = 0
        self._total_loss = 0
        
        # 가용성 기록 (timestamp, is_up)
        self._availability_records: deque = deque()
        self._last_up_check: Optional[datetime] = None
    
    def record_packet(
        self,
        timestamp: datetime,
        sequence: int,
        size: int = 0
    ) -> Optional[float]:
        """
        패킷 수신 기록 및 지터 계산.
        
        Args:
            timestamp: 수신 시간
            sequence: 시퀀스 번호 (0-255)
            size: 패킷 크기 (바이트)
            
        Returns:
            계산된 지터 값 (밀리초) 또는 None
        """
        with self._lock:
            # 패킷 기록 추가
            self._packets.append(PacketRecord(timestamp, sequence, size))
            self._total_packets += 1
            
            # 손실 추정
            self._estimate_loss(sequence)
            
            # 지터 계산
            jitter = self._calculate_jitter(timestamp)
            if jitter is not None:
                self._jitter_buffer.append(jitter)
            
            # 이전 패킷 정보 업데이트
            self._prev_packet_time = timestamp
            self._prev_sequence = sequence
            
            # 오래된 기록 정리
            self._prune_old_records(timestamp)
            
            # 가용성 기록
            self._record_availability(timestamp, is_up=True)
            
            return jitter
    
    def _estimate_loss(self, current_seq: int) -> int:
        """
        시퀀스 기반 손실 추정.
        
        시퀀스 번호 갭을 기반으로 손실된 패킷 수를 추정합니다.
        0-255 롤오버를 고려합니다.
        
        Args:
            current_seq: 현재 시퀀스 번호
            
        Returns:
            추정 손실 수
        
        Q: 왜 롤오버 처리가 필요한가요?
        A: 8비트 시퀀스 번호는 0부터 255까지 반복됩니다. 255 다음에 0이 오는 경우를
           정상적인 다음 패킷으로 인식하기 위해 특별한 처리가 필요합니다.
           예: prev=255, curr=0 이면 갭은 1입니다.
        
        Q: 왜 큰 갭(100 이상)은 재시작으로 간주하나요?
        A: 실제로 100개 이상의 패킷이 연속으로 손실되는 경우는 거의 없습니다.
           큰 갭은 시스템 재시작이나 네트워크 재연결로 인한 것으로 간주하여
           손실로 카운트하지 않습니다.
        """
        if self._prev_sequence is None:
            return 0
        
        expected = (self._prev_sequence + 1) % 256  # 롤오버 고려
        
        if current_seq == expected:
            return 0
        
        # 갭 계산 (롤오버 고려)
        if current_seq > expected:
            gap = current_seq - expected
        else:
            # 롤오버 발생 (예: 255 -> 0)
            gap = (256 - expected) + current_seq
        
        # 큰 갭(100 이상)은 재시작으로 간주
        if gap < 100:
            self._total_loss += gap
            return gap
        
        return 0
    
    def _calculate_jitter(self, current_time: datetime) -> Optional[float]:
        """
        지터 계산.
        
        RFC 3550 방식: 이전 패킷과의 도착 시간 차이 변동
        
        Args:
            current_time: 현재 패킷 도착 시간
            
        Returns:
            지터 값 (밀리초) 또는 None
        
        Q: 지터는 무엇인가요?
        A: 지터(Jitter)는 패킷 도착 시간의 변동성을 나타냅니다.
           네트워크 품질 지표로 사용되며, 낮을수록 안정적인 통신을 의미합니다.
           예: 1ms 간격으로 패킷이 도착해야 하는데, 0.5ms, 1.5ms, 0.8ms로 변동하면 지터가 발생합니다.
        
        Q: 왜 P95/P99 백분위수를 사용하나요?
        A: 평균 지터는 일시적인 큰 변동을 완화하지만, 최악의 경우를 파악하기 어렵습니다.
           P95 지터는 전체 지터 값 중 95%가 이 값보다 작거나 같다는 의미로,
           네트워크 품질의 현실적인 지표를 제공합니다.
        """
        if self._prev_packet_time is None:
            return None
        
        # 도착 간격 (밀리초)
        delta_ms = (current_time - self._prev_packet_time).total_seconds() * 1000
        
        # 지터 = 현재 간격 - 평균 간격의 절대값
        # 간단히 간격 자체를 지터로 사용 (1ms 기준)
        expected_interval = 1.0  # 1ms (1000 pps 가정)
        jitter = abs(delta_ms - expected_interval)
        
        return jitter
    
    def _prune_old_records(self, now: datetime) -> None:
        """오래된 기록 제거."""
        cutoff = now - timedelta(seconds=self._window_sec)
        
        # 패킷 기록 정리
        while self._packets and self._packets[0].timestamp < cutoff:
            self._packets.popleft()
        
        # 가용성 기록 정리
        while (self._availability_records and 
               self._availability_records[0][0] < cutoff):
            self._availability_records.popleft()
    
    def _record_availability(self, timestamp: datetime, is_up: bool) -> None:
        """가용성 상태 기록."""
        self._availability_records.append((timestamp, is_up))
        self._last_up_check = timestamp
    
    def record_down(self, timestamp: datetime) -> None:
        """
        다운 상태 기록.
        
        패킷이 일정 시간 동안 수신되지 않을 때 호출합니다.
        """
        with self._lock:
            self._record_availability(timestamp, is_up=False)
    
    def get_pps(self) -> int:
        """
        현재 PPS 계산.
        
        최근 1초간의 패킷 수를 반환합니다.
        
        Returns:
            초당 패킷 수
        """
        with self._lock:
            if not self._packets:
                return 0
            
            now = datetime.now()
            one_sec_ago = now - timedelta(seconds=1)
            
            count = sum(1 for p in self._packets if p.timestamp >= one_sec_ago)
            return count
    
    def get_jitter_percentiles(self) -> Tuple[float, float]:
        """
        지터 P95, P99 계산.
        
        Returns:
            (P95, P99) 튜플 (밀리초)
        """
        with self._lock:
            if not self._jitter_buffer:
                return (0.0, 0.0)
            
            sorted_values = sorted(self._jitter_buffer)
            n = len(sorted_values)
            
            # P95
            idx_95 = int(n * 0.95)
            p95 = sorted_values[min(idx_95, n - 1)]
            
            # P99
            idx_99 = int(n * 0.99)
            p99 = sorted_values[min(idx_99, n - 1)]
            
            return (round(p95, 2), round(p99, 2))
    
    def get_packet_loss(self) -> int:
        """
        추정 패킷 손실 수.
        
        Returns:
            손실된 패킷 수
        """
        with self._lock:
            return self._total_loss
    
    def get_availability(self, window_sec: int = 300) -> float:
        """
        가용성 계산.
        
        지정된 시간 윈도우 내의 가용성 백분율을 계산합니다.
        
        Args:
            window_sec: 시간 윈도우 (초, 기본값 300 = 5분)
            
        Returns:
            가용성 백분율 (0-100)
        """
        with self._lock:
            if not self._availability_records:
                return 100.0
            
            now = datetime.now()
            cutoff = now - timedelta(seconds=window_sec)
            
            # 윈도우 내 기록 필터링
            records = [
                (ts, up) for ts, up in self._availability_records
                if ts >= cutoff
            ]
            
            if not records:
                return 100.0
            
            # Up 시간 계산
            up_time = 0.0
            for i, (ts, is_up) in enumerate(records):
                if not is_up:
                    continue
                
                # 다음 기록까지의 시간 (또는 현재까지)
                if i + 1 < len(records):
                    next_ts = records[i + 1][0]
                else:
                    next_ts = now
                
                up_time += (next_ts - ts).total_seconds()
            
            total_time = min(
                (now - records[0][0]).total_seconds(),
                window_sec
            )
            
            if total_time <= 0:
                return 100.0
            
            return round((up_time / total_time) * 100, 2)
    
    def reset(self) -> None:
        """통계 초기화."""
        with self._lock:
            self._packets.clear()
            self._jitter_buffer.clear()
            self._availability_records.clear()
            self._prev_packet_time = None
            self._prev_sequence = None
            self._total_packets = 0
            self._total_loss = 0
```

```python
"""
Anomaly Detector Module.

통신 품질 이상 탐지 규칙을 정의하고 적용합니다.

이상 유형:
- HIGH_JITTER: 지터 임계값 초과
- PACKET_LOSS: 패킷 손실 감지
- CONNECTION_TIMEOUT: 연결 타임아웃
- CHECKSUM_FAILURE: 체크섬 실패
- LOW_AVAILABILITY: 가용성 저하
"""

from dataclasses import dataclass
from datetime import datetime
from enum import Enum, auto
from typing import List, Optional

from ..config import config


class AnomalyType(Enum):
    """이상 유형."""
    HIGH_JITTER = auto()        # 높은 지터
    PACKET_LOSS = auto()        # 패킷 손실
    CONNECTION_TIMEOUT = auto() # 연결 타임아웃
    CHECKSUM_FAILURE = auto()   # 체크섬 실패
    LOW_AVAILABILITY = auto()   # 낮은 가용성
    PARSE_FAILURE = auto()      # 파싱 실패


class AnomalySeverity(Enum):
    """이상 심각도."""
    INFO = "info"           # 정보
    WARNING = "warning"     # 경고
    CRITICAL = "critical"   # 심각


@dataclass
class Anomaly:
    """
    탐지된 이상.
    
    Attributes:
        type: 이상 유형
        severity: 심각도
        message: 상세 메시지
        timestamp: 발생 시간
        value: 관련 값 (옵션)
        threshold: 임계값 (옵션)
    """
    type: AnomalyType
    severity: AnomalySeverity
    message: str
    timestamp: datetime
    value: Optional[float] = None
    threshold: Optional[float] = None
    
    def to_dict(self) -> dict:
        """딕셔너리 변환."""
        return {
            "type": self.type.name,
            "severity": self.severity.value,
            "message": self.message,
            "timestamp": self.timestamp.strftime("%H:%M:%S"),
            "value": self.value,
            "threshold": self.threshold,
        }


class AnomalyDetector:
    """
    이상 탐지기.
    
    통계값을 기반으로 이상을 탐지합니다.
    임계값은 config에서 로드합니다.
    
    Usage:
        detector = AnomalyDetector()
        
        # 지터 검사
        anomalies = detector.check_jitter(current_jitter=15.5)
        
        # 연결 상태 검사
        anomalies = detector.check_connection(last_packet_time)
    """
    
    def __init__(self):
        """AnomalyDetector 초기화."""
        # 임계값 로드
        self._jitter_warning = config.threshold.jitter_warning_ms
        self._jitter_critical = config.threshold.jitter_critical_ms
        self._loss_warning = config.threshold.packet_loss_warning
        self._availability_warning = config.threshold.availability_warning
        self._availability_critical = config.threshold.availability_critical
        self._checksum_warning = config.threshold.checksum_fail_warning
        self._down_threshold = config.ui.down_threshold_sec
    
    def check_jitter(self, jitter_p95: float, jitter_p99: float) -> List[Anomaly]:
        """
        지터 임계값 검사.
        
        Args:
            jitter_p95: P95 지터 (ms)
            jitter_p99: P99 지터 (ms)
            
        Returns:
            탐지된 이상 리스트
        """
        anomalies = []
        now = datetime.now()
        
        # P99 기준으로 검사
        if jitter_p99 >= self._jitter_critical:
            anomalies.append(Anomaly(
                type=AnomalyType.HIGH_JITTER,
                severity=AnomalySeverity.CRITICAL,
                message=f"지터 심각: P99={jitter_p99:.1f}ms (임계값: {self._jitter_critical}ms)",
                timestamp=now,
                value=jitter_p99,
                threshold=self._jitter_critical,
            ))
        elif jitter_p99 >= self._jitter_warning:
            anomalies.append(Anomaly(
                type=AnomalyType.HIGH_JITTER,
                severity=AnomalySeverity.WARNING,
                message=f"지터 경고: P99={jitter_p99:.1f}ms (임계값: {self._jitter_warning}ms)",
                timestamp=now,
                value=jitter_p99,
                threshold=self._jitter_warning,
            ))
        
        return anomalies
    
    def check_packet_loss(self, loss_count: int) -> List[Anomaly]:
        """
        패킷 손실 검사.
        
        Args:
            loss_count: 추정 손실 패킷 수
            
        Returns:
            탐지된 이상 리스트
        """
        anomalies = []
        now = datetime.now()
        
        if loss_count >= self._loss_warning:
            anomalies.append(Anomaly(
                type=AnomalyType.PACKET_LOSS,
                severity=AnomalySeverity.WARNING,
                message=f"패킷 손실 감지: {loss_count}개 (임계값: {self._loss_warning}개)",
                timestamp=now,
                value=float(loss_count),
                threshold=float(self._loss_warning),
            ))
        
        return anomalies
    
    def check_connection(
        self,
        last_packet_time: Optional[datetime]
    ) -> List[Anomaly]:
        """
        연결 상태 검사.
        
        Args:
            last_packet_time: 마지막 패킷 수신 시간
            
        Returns:
            탐지된 이상 리스트
        """
        anomalies = []
        now = datetime.now()
        
        if last_packet_time is None:
            anomalies.append(Anomaly(
                type=AnomalyType.CONNECTION_TIMEOUT,
                severity=AnomalySeverity.CRITICAL,
                message="패킷 수신 없음",
                timestamp=now,
            ))
            return anomalies
        
        elapsed = (now - last_packet_time).total_seconds()
        
        if elapsed >= self._down_threshold:
            anomalies.append(Anomaly(
                type=AnomalyType.CONNECTION_TIMEOUT,
                severity=AnomalySeverity.CRITICAL,
                message=f"연결 타임아웃: {elapsed:.1f}초 동안 패킷 없음",
                timestamp=now,
                value=elapsed,
                threshold=self._down_threshold,
            ))
        
        return anomalies
    
    def check_availability(self, availability_pct: float) -> List[Anomaly]:
        """
        가용성 검사.
        
        Args:
            availability_pct: 가용성 백분율
            
        Returns:
            탐지된 이상 리스트
        """
        anomalies = []
        now = datetime.now()
        
        if availability_pct < self._availability_critical:
            anomalies.append(Anomaly(
                type=AnomalyType.LOW_AVAILABILITY,
                severity=AnomalySeverity.CRITICAL,
                message=f"가용성 심각: {availability_pct:.1f}% (임계값: {self._availability_critical}%)",
                timestamp=now,
                value=availability_pct,
                threshold=self._availability_critical,
            ))
        elif availability_pct < self._availability_warning:
            anomalies.append(Anomaly(
                type=AnomalyType.LOW_AVAILABILITY,
                severity=AnomalySeverity.WARNING,
                message=f"가용성 경고: {availability_pct:.1f}% (임계값: {self._availability_warning}%)",
                timestamp=now,
                value=availability_pct,
                threshold=self._availability_warning,
            ))
        
        return anomalies
    
    def check_checksum(self, fail_pct: float) -> List[Anomaly]:
        """
        체크섬 실패율 검사.
        
        Args:
            fail_pct: 체크섬 실패율 (%)
            
        Returns:
            탐지된 이상 리스트
        """
        anomalies = []
        now = datetime.now()
        
        if fail_pct >= self._checksum_warning:
            anomalies.append(Anomaly(
                type=AnomalyType.CHECKSUM_FAILURE,
                severity=AnomalySeverity.WARNING,
                message=f"체크섬 실패 증가: {fail_pct:.1f}% (임계값: {self._checksum_warning}%)",
                timestamp=now,
                value=fail_pct,
                threshold=self._checksum_warning,
            ))
        
        return anomalies
    
    def check_all(
        self,
        jitter_p95: float = 0,
        jitter_p99: float = 0,
        loss_count: int = 0,
        last_packet_time: Optional[datetime] = None,
        availability_pct: float = 100,
        checksum_fail_pct: float = 0,
    ) -> List[Anomaly]:
        """
        모든 이상 검사 수행.
        
        Returns:
            탐지된 모든 이상 리스트
        """
        anomalies = []
        
        anomalies.extend(self.check_jitter(jitter_p95, jitter_p99))
        anomalies.extend(self.check_packet_loss(loss_count))
        anomalies.extend(self.check_connection(last_packet_time))
        anomalies.extend(self.check_availability(availability_pct))
        anomalies.extend(self.check_checksum(checksum_fail_pct))
        
        return anomalies
```
