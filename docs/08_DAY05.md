## 💻 코드 구현 내용

### 1. `ugv_mon/analysis/availability.py` - 가용성 분석기

**역할**: 패킷 수신 여부를 기반으로 통신 가용성 계산 및 Down 구간 탐지

**구현 내용**:

```python
"""
가용성 분석 모듈

역할:
- 패킷 수신 여부 기반 Up/Down 판정
- Down 임계값 설정 (기본 5초)
- 5분/1시간 롤링 가용성 계산
- 타임라인 세그먼트 생성
"""

from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import List, Optional, Dict

from ..config import config


@dataclass
class AvailabilitySegment:
    """
    가용성 타임라인 세그먼트
    
    Represents a continuous period of uptime or downtime.
    
    Attributes:
        start_time: 시작 시간
        end_time: 종료 시간
        is_up: True = uptime, False = downtime
    """
    start_time: datetime
    end_time: datetime
    is_up: bool
    
    @property
    def duration_sec(self) -> float:
        """세그먼트 지속 시간 (초)"""
        return (self.end_time - self.start_time).total_seconds()
    
    def to_dict(self) -> Dict:
        """
        대시보드용 딕셔너리 변환
        
        타임라인 차트에서 사용할 수 있도록 상대 시간(초)으로 변환
        """
        # 타임라인 시작 시점을 기준으로 상대 시간 계산
        # (실제 구현에서는 타임라인 윈도우 시작 시간 필요)
        return {
            "start": self.start_time.timestamp(),
            "end": self.end_time.timestamp(),
            "status": "up" if self.is_up else "down",
        }


class AvailabilityAnalyzer:
    """
    가용성 분석기
    
    패킷 수신 여부를 추적하여 시스템 가용성을 계산합니다.
    지정된 시간 동안 패킷이 없으면 Down으로 판정합니다.
    
    사용법:
        analyzer = AvailabilityAnalyzer(down_threshold_sec=5.0)
        analyzer.record_packet(datetime.now())
        availability = analyzer.get_availability(window_sec=300)  # 5분
    """
    
    def __init__(
        self,
        down_threshold_sec: float = None,
        window_5min: int = None,
        window_1hour: int = None,
    ):
        """
        초기화
        
        Args:
            down_threshold_sec: Down 판정 임계값 (초), 기본값: config에서 가져옴
            window_5min: 5분 윈도우 (초)
            window_1hour: 1시간 윈도우 (초)
        """
        self.down_threshold_sec = down_threshold_sec or config.ui.down_threshold_sec
        self.window_5min = window_5min or config.ui.availability_window_5min
        self.window_1hour = window_1hour or config.ui.availability_window_1hour
        
        self._last_packet_time: Optional[datetime] = None
        self._segments: List[AvailabilitySegment] = []
        self._current_segment: Optional[AvailabilitySegment] = None
        self._segment_start_time: Optional[datetime] = None
        
    def record_packet(self, timestamp: datetime) -> None:
        """
        패킷 수신 기록
        
        Args:
            timestamp: 패킷 수신 시간
        """
        now = timestamp
        
        if self._last_packet_time is None:
            # 첫 패킷 - Up 세그먼트 시작
            self._start_segment(now, is_up=True)
        else:
            # 마지막 패킷 이후 경과 시간
            gap = (now - self._last_packet_time).total_seconds()
            
            if gap > self.down_threshold_sec:
                # Down 구간 발생
                # 현재 세그먼트 종료
                self._end_segment(self._last_packet_time)
                
                # Down 세그먼트 생성
                self._start_segment(self._last_packet_time, is_up=False)
                self._end_segment(now)
                
                # 새 Up 세그먼트 시작
                self._start_segment(now, is_up=True)
            else:
                # 정상 - 현재 세그먼트 연장
                if self._current_segment:
                    self._current_segment.end_time = now
        
        self._last_packet_time = now
    
    def _start_segment(self, time: datetime, is_up: bool) -> None:
        """
        새 세그먼트 시작
        
        Args:
            time: 세그먼트 시작 시간
            is_up: True = Up, False = Down
        """
        self._current_segment = AvailabilitySegment(
            start_time=time,
            end_time=time,
            is_up=is_up,
        )
        self._segment_start_time = time
    
    def _end_segment(self, time: datetime) -> None:
        """
        현재 세그먼트 종료
        
        Args:
            time: 세그먼트 종료 시간
        """
        if self._current_segment:
            self._current_segment.end_time = time
            self._segments.append(self._current_segment)
            self._current_segment = None
    
    def get_availability(self, window_sec: int) -> float:
        """
        지정 윈도우의 가용성 계산
        
        Args:
            window_sec: 시간 윈도우 (초)
            
        Returns:
            가용성 퍼센트 (0-100)
        """
        now = datetime.now()
        cutoff = now - timedelta(seconds=window_sec)
        
        up_time = 0.0
        total_time = 0.0
        
        # 세그먼트들을 순회하며 Up 시간 계산
        for seg in self._segments:
            if seg.end_time < cutoff:
                continue
            
            # 윈도우와 겹치는 부분만 계산
            start = max(seg.start_time, cutoff)
            end = min(seg.end_time, now)
            duration = (end - start).total_seconds()
            
            total_time += duration
            if seg.is_up:
                up_time += duration
        
        # 현재 세그먼트 처리
        if self._current_segment and self._current_segment.start_time >= cutoff:
            start = max(self._current_segment.start_time, cutoff)
            end = now
            duration = (end - start).total_seconds()
            
            total_time += duration
            if self._current_segment.is_up:
                up_time += duration
        
        if total_time == 0:
            return 100.0  # 데이터 없으면 100%로 간주
        
        return (up_time / total_time) * 100.0
    
    def get_metrics(self) -> Dict:
        """
        대시보드용 메트릭 반환
        
        Returns:
            메트릭 딕셔너리
        """
        return {
            "availability5min": round(self.get_availability(self.window_5min), 2),
            "availability1hour": round(self.get_availability(self.window_1hour), 2),
        }
    
    def get_timeline_segments(self, window_sec: int = None) -> List[Dict]:
        """
        타임라인 차트용 세그먼트 반환
        
        Args:
            window_sec: 타임라인 윈도우 (초), 기본값: config에서 가져옴
            
        Returns:
            세그먼트 딕셔너리 리스트 (상대 시간 기준)
        """
        if window_sec is None:
            window_sec = config.ui.timeline_duration_sec
        
        now = datetime.now()
        cutoff = now - timedelta(seconds=window_sec)
        
        result = []
        
        # 과거 세그먼트들
        for seg in self._segments:
            if seg.end_time < cutoff:
                continue
            
            # 상대 시간으로 변환 (0 = window_sec 전, window_sec = 현재)
            start_rel = max(0, (seg.start_time - cutoff).total_seconds())
            end_rel = min(window_sec, (seg.end_time - cutoff).total_seconds())
            
            if end_rel > start_rel:  # 유효한 세그먼트만
                result.append({
                    "start": start_rel,
                    "end": end_rel,
                    "status": "up" if seg.is_up else "down",
                })
        
        # 현재 세그먼트
        if self._current_segment and self._current_segment.start_time >= cutoff:
            start_rel = max(0, (self._current_segment.start_time - cutoff).total_seconds())
            end_rel = window_sec
            
            if end_rel > start_rel:
                result.append({
                    "start": start_rel,
                    "end": end_rel,
                    "status": "up" if self._current_segment.is_up else "down",
                })
        
        return result
    
    def check_down_status(self) -> bool:
        """
        현재 Down 상태인지 확인
        
        Returns:
            True if currently down (no packet for threshold time)
        """
        if self._last_packet_time is None:
            return True  # 패킷 없으면 Down
        
        elapsed = (datetime.now() - self._last_packet_time).total_seconds()
        return elapsed > self.down_threshold_sec
```

**설명 및 근거**:
- **Down 임계값**: 5초 이상 패킷 없으면 Down 판정 (configurable)
- **세그먼트 기반 추적**: 연속된 Up/Down 구간을 세그먼트로 관리
- **롤링 윈도우**: 5분/1시간 윈도우에서 가용성 계산
- **타임라인 지원**: 차트 표시를 위한 상대 시간 변환

```python
# test_availability.py
from ugv_mon.analysis.availability import AvailabilityAnalyzer
from datetime import datetime, timedelta

def test_availability_calculation():
    """가용성 계산 테스트"""
    analyzer = AvailabilityAnalyzer(down_threshold_sec=5.0)
    
    now = datetime.now()
    
    # 정상 패킷 (1초 간격, 10개)
    for i in range(10):
        analyzer.record_packet(now + timedelta(seconds=i))
    
    # 10초 갭 (Down 발생)
    analyzer.record_packet(now + timedelta(seconds=20))
    
    # 다시 정상 (5개)
    for i in range(5):
        analyzer.record_packet(now + timedelta(seconds=21+i))
    
    availability = analyzer.get_availability(window_sec=30)
    print(f"Availability: {availability:.2f}%")
    assert availability < 100.0  # Down 구간이 있으므로 100% 미만

def test_down_segment_detection():
    """Down 세그먼트 탐지 테스트"""
    analyzer = AvailabilityAnalyzer(down_threshold_sec=5.0)
    
    now = datetime.now()
    
    # 정상 → Down → 정상
    analyzer.record_packet(now)
    analyzer.record_packet(now + timedelta(seconds=20))  # 20초 후
    
    segments = analyzer.get_timeline_segments(window_sec=30)
    
    # Up → Down → Up 순서 확인
    assert len(segments) >= 2
```
