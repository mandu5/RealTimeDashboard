## 💻 코드 구현 내용

### 1. `ugv_mon/data/live_data.py` - 실시간 데이터 제공자

**역할**: 캡처/파싱/분석 모듈을 통합하여 대시보드에 실시간 데이터 제공

**구현 내용**:

```python
"""
실시간 데이터 제공자

역할:
- 캡처/파싱/분석 모듈 통합
- 대시보드용 데이터 딕셔너리 생성
- 로그 엔트리 관리
- 차트 데이터 수집
"""

from datetime import datetime
from typing import Dict, List
from collections import deque
from threading import Thread, Lock
import time

from ..capture.sniffer import PacketSniffer
from ..parser.icd_parser import ICDParser, ParseResult
from ..analysis.quality import QualityAnalyzer
from ..analysis.availability import AvailabilityAnalyzer
from ..analysis.jitter import JitterAnalyzer
from ..analysis.loss import LossEstimator
from ..utils.helpers import (
    get_operation_mode_label,
    get_authority_label,
    get_driving_state_label,
)
from ..config import config


class LiveDataProvider:
    """
    실시간 데이터 제공자
    
    모든 분석 모듈을 통합하여 대시보드에 실시간 데이터를 제공합니다.
    별도 스레드에서 패킷 처리 루프를 실행합니다.
    
    사용법:
        provider = LiveDataProvider()
        provider.start()
        data = provider.get_dashboard_data()
        provider.stop()
    """
    
    def __init__(self):
        """초기화 - 모든 모듈 인스턴스 생성"""
        # 모듈 초기화
        self.sniffer = PacketSniffer(interface=config.network.interface)
        self.parser = ICDParser()
        self.quality = QualityAnalyzer()
        self.availability = AvailabilityAnalyzer()
        self.jitter = JitterAnalyzer()
        self.loss = LossEstimator()
        
        # 상태 저장
        self._lock = Lock()
        self._latest_state: Dict = {}
        self._logs: deque = deque(maxlen=config.ui.max_log_entries)
        self._chart_data: deque = deque(maxlen=config.ui.max_chart_points)
        
        # 처리 스레드
        self._running = False
        self._thread: Thread = None
        
        # 초기 상태 설정
        self._latest_state = {
            "connected": False,
            "interface": config.network.interface,
            "filter": f"{config.network.source_port}→{config.network.dest_port}",
            "operationalMode": "준비 (PREP)",
            "operationalAuthority": "해제됨 (RELEASED)",
            "drivingState": "원격 (REMOTE)",
            "devices": [],
            "emergencyStatus": {},
        }
        
    def start(self) -> None:
        """데이터 수집 시작"""
        if self._running:
            return
        
        self.sniffer.start()
        self._running = True
        self._thread = Thread(target=self._process_loop, daemon=True)
        self._thread.start()
    
    def stop(self) -> None:
        """데이터 수집 중지"""
        self._running = False
        self.sniffer.stop()
        if self._thread:
            self._thread.join(timeout=2.0)
    
    def _process_loop(self) -> None:
        """패킷 처리 루프 (별도 스레드에서 실행)"""
        while self._running:
            packet = self.sniffer.get_packet(timeout=0.1)
            if packet is None:
                continue
            
            now = datetime.now()
            
            try:
                # UDP 페이로드 추출
                payload = bytes(packet['UDP'].payload)
            except Exception:
                continue
            
            # 파싱
            result = self.parser.parse(payload)
            
            # 분석 업데이트
            self.quality.record(result)
            self.availability.record_packet(now)
            self.jitter.record_packet(now)
            
            if result.header:
                self.loss.record_sequence(result.header.sequence)
            
            # 상태 업데이트
            with self._lock:
                self._update_state(result, now)
                self._add_log_entry(result, now)
                self._update_chart_data(now)
    
    def _update_state(self, result: ParseResult, now: datetime) -> None:
        """상태 업데이트"""
        self._latest_state["connected"] = True
        self._latest_state["lastPacketTime"] = now.strftime("%H:%M:%S")
        
        if result.success and result.payload:
            p = result.payload
            
            # 운용 상태
            self._latest_state["operationalMode"] = get_operation_mode_label(p.operation_mode)
            self._latest_state["operationalAuthority"] = get_authority_label(p.authority)
            self._latest_state["drivingState"] = get_driving_state_label(p.driving_state)
            
            # 장치 연결
            self._latest_state["devices"] = self._parse_devices(p.device_presence)
            
            # 비상정지 원인
            self._latest_state["emergencyStatus"] = self._parse_emergency(p.emergency_bits)
    
    def _parse_devices(self, bits: int) -> List[Dict]:
        """장치 연결 비트 파싱"""
        devices = []
        device_names = ["VIC", "RDC", "ADC", "FCAM", "RCAM", 
                       "AUX", "SCS", "DIP", "TCC", "TM"]
        
        for i, name in enumerate(device_names):
            connected = bool(bits & (1 << i))
            devices.append({
                "name": name,
                "connected": connected,
                "warning": False,  # 추후 경고 조건 추가 가능
            })
        
        return devices
    
    def _parse_emergency(self, bits: int) -> Dict[str, bool]:
        """비상정지 비트 파싱"""
        names = [
            "통신 두절",           # bit 6
            "장비고장(주행)",      # bit 7
            "장비고장(동력계)",    # bit 8
            "신호단절(주행)",      # bit 9
            "신호단절(자율)",      # bit 10
            "신호단절(항법)",      # bit 11
            "신호단절(동력계)",    # bit 12
            "신호단절(통신)",      # bit 13
            "수동정지(운용통제장치)",  # bit 14
            "수동정지(근거리조종기)",  # bit 15
        ]
        
        result = {}
        for i, name in enumerate(names):
            result[name] = bool(bits & (1 << i))
        
        return result
    
    def _add_log_entry(self, result: ParseResult, now: datetime) -> None:
        """로그 엔트리 추가"""
        entry = {
            "time": now.strftime("%H:%M:%S"),  # HH:MM:SS (밀리초 제거)
            "seq": result.header.sequence if result.header else 0,
            "msg_code": f"0x{result.header.msg_code:02X}" if result.header else "---",
            "parse_ok": "✓" if result.success else "✗",
            "checksum_ok": "✓" if result.checksum_ok else "✗",
            "mode": self._latest_state.get("operationalMode", "---"),
            "authority": self._latest_state.get("operationalAuthority", "---"),
            "notes": result.error if not result.success else "",
        }
        self._logs.appendleft(entry)
    
    def _update_chart_data(self, now: datetime) -> None:
        """차트 데이터 포인트 추가"""
        stats = self.sniffer.get_stats()
        jitter_metrics = self.jitter.get_metrics()
        
        point = {
            "timestamp": now.strftime("%H:%M:%S"),
            "pps": stats.get("total_filtered", 0),
            "jitter": jitter_metrics.get("jitterP95", 0),
        }
        
        self._chart_data.append(point)
    
    def get_dashboard_data(self) -> Dict:
        """대시보드용 전체 데이터 반환"""
        with self._lock:
            stats = self.sniffer.get_stats()
            
            return {
                **self._latest_state,
                "capturePps": stats.get("total_filtered", 0),
                "filterPass": 100.0,  # Scapy 필터링은 100%
                **self.quality.get_metrics_dict(),
                **self.availability.get_metrics(),
                **self.jitter.get_metrics(),
                **self.loss.get_metrics(),
                "availabilitySegments": self.availability.get_timeline_segments(),
                "combinedData": list(self._chart_data),
            }
    
    def get_logs(self, limit: int = 50) -> List[Dict]:
        """로그 엔트리 반환"""
        with self._lock:
            return list(self._logs)[:limit]
    
    @property
    def log_count(self) -> int:
        """현재 로그 엔트리 수"""
        return len(self._logs)
    
    def clear_logs(self) -> None:
        """로그 삭제"""
        with self._lock:
            self._logs.clear()
```

**설명 및 근거**:
- **스레드 분리**: 패킷 처리는 별도 스레드에서 실행하여 UI 블로킹 방지
- **Lock 사용**: 멀티스레드 환경에서 상태 동기화
- **deque 사용**: 로그/차트 데이터의 롤링 윈도우 구현
- **모듈 통합**: 캡처→파싱→분석 파이프라인 통합
