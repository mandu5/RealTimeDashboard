## 💻 코드 구현

### 1. `ugv_mon/data/live_provider.py`

> 캡처된 패킷을 파싱하고 UI에 전달할 데이터를 생성하는 클래스입니다.

```python
"""
Live Data Provider Module.

실시간 패킷 캡처와 ICD 파싱을 통합하여
대시보드 UI에 데이터를 제공합니다.

데이터 흐름:
1. PacketSniffer가 UDP 패킷 캡처
2. 캡처된 패킷을 PacketQueue에 저장
3. UI 폴링 시 큐에서 패킷을 꺼내 ICDParser로 파싱
4. 파싱 결과를 기반으로 대시보드 상태 업데이트

MockDataGenerator와 동일한 인터페이스를 제공하여
개발 중에는 Mock, 실제 환경에서는 Live로 전환 가능합니다.
"""

import logging
from collections import deque
from datetime import datetime
from typing import Dict, List, Optional
from threading import Lock

from ..capture import PacketSniffer, PacketQueue, CaptureStats
from ..parser import ICDParser, ParseResult
from ..config import config
from .models import LogEntry, DeviceStatus, AvailabilitySegment, EmergencyStatus

# 로거 설정
logger = logging.getLogger(__name__)


class LiveDataProvider:
    """
    실시간 데이터 제공자.
    
    PacketSniffer와 ICDParser를 연결하여 대시보드에
    실시간 데이터를 제공합니다.
    
    MockDataGenerator와 동일한 메서드 시그니처를 제공하여
    코드 변경 없이 데이터 소스를 전환할 수 있습니다.
    
    Usage:
        provider = LiveDataProvider(interface="lo")
        provider.start_capture()
        
        # 폴링 시마다 호출
        data = provider.get_dashboard_data()
        logs = provider.get_logs()
    """
    
    def __init__(
        self,
        interface: str = None,
        src_port: int = None,
        dst_port: int = None
    ):
        """
        LiveDataProvider 초기화.
        
        Args:
            interface: 캡처 인터페이스 (기본값: config에서 로드)
            src_port: 송신 포트 (기본값: config에서 로드)
            dst_port: 수신 포트 (기본값: config에서 로드)
        """
        # 설정값 로드 (인자가 없으면 config 사용)
        self._interface = interface or config.network.interface
        self._src_port = src_port or config.network.source_port
        self._dst_port = dst_port or config.network.dest_port
        
        # 캡처 컴포넌트
        self._packet_queue = PacketQueue(max_size=1000)
        self._sniffer: Optional[PacketSniffer] = None
        
        # 파서
        self._parser = ICDParser()
        
        # 상태 데이터 (스레드 안전을 위한 Lock)
        self._lock = Lock()
        self._connected = False
        self._last_packet_time: Optional[datetime] = None
        
        # 통계
        self._parse_total = 0
        self._parse_success = 0
        self._checksum_fail = 0
        self._last_seq: Optional[int] = None
        self._packet_loss = 0
        
        # 지터 계산용 버퍼
        self._packet_times: deque = deque(maxlen=100)
        self._jitter_values: deque = deque(maxlen=100)
        
        # 로그 버퍼 (최근 200개)
        self._logs: deque = deque(maxlen=config.ui.max_log_entries)
        
        # 차트 데이터 버퍼 (최근 60개)
        self._chart_data: deque = deque(maxlen=config.ui.max_chart_points)
        
        # 가용성 세그먼트
        self._availability_segments: List[AvailabilitySegment] = []
        self._elapsed_time = 0
        
        # 최신 파싱 상태 (운용상태, 장치상태 등)
        self._latest_state: Dict = {}
        
        logger.info(
            f"LiveDataProvider initialized: {self._interface}, "
            f"{self._src_port}→{self._dst_port}"
        )
    
    def start_capture(self) -> bool:
        """
        패킷 캡처 시작.
        
        PacketSniffer를 생성하고 캡처를 시작합니다.
        root 권한이 필요합니다.
        
        Returns:
            성공 여부
        """
        try:
            self._sniffer = PacketSniffer(
                interface=self._interface,
                src_port=self._src_port,
                dst_port=self._dst_port,
                callback=self._packet_queue.put  # 콜백으로 큐에 저장
            )
            self._sniffer.start()
            logger.info("Capture started")
            return True
            
        except ImportError as e:
            logger.error(f"Scapy not available: {e}")
            return False
        except PermissionError:
            logger.error("Permission denied. Run with sudo.")
            return False
        except Exception as e:
            logger.error(f"Failed to start capture: {e}")
            return False
    
    def stop_capture(self) -> None:
        """패킷 캡처 중지."""
        if self._sniffer:
            self._sniffer.stop()
            logger.info("Capture stopped")
    
    def is_capturing(self) -> bool:
        """캡처 실행 중 여부."""
        return self._sniffer is not None and self._sniffer.is_running()
    
    def process_pending_packets(self) -> int:
        """
        큐에 대기 중인 패킷 처리.
        
        PacketQueue에서 모든 패킷을 꺼내 파싱합니다.
        UI 폴링 콜백에서 호출됩니다.
        
        Returns:
            처리된 패킷 수
        """
        packets = self._packet_queue.get_all()
        
        for raw_data in packets:
            self._process_single_packet(raw_data)
        
        return len(packets)
    
    def _process_single_packet(self, raw_data: bytes) -> None:
        """
        단일 패킷 처리.
        
        파싱 → 통계 업데이트 → 로그 추가 → 상태 업데이트
        
        Args:
            raw_data: 원시 패킷 바이트
        """
        now = datetime.now()
        
        with self._lock:
            # 연결 상태 업데이트
            self._connected = True
            self._last_packet_time = now
            
            # 파싱
            self._parse_total += 1
            result = self._parser.parse(raw_data)
            
            if result.success:
                self._parse_success += 1
                
                if not result.checksum_ok:
                    self._checksum_fail += 1
                
                # 시퀀스 기반 손실 추정
                if result.header:
                    self._update_loss_estimation(result.header.sequence)
                
                # 지터 계산
                self._update_jitter(now)
                
                # 상태 업데이트 (상태보고 메시지인 경우)
                if result.is_status_report() and result.payload:
                    self._update_state_from_payload(result.payload)
            
            # 로그 추가
            self._add_log_entry(result, now)
    
    def _update_loss_estimation(self, current_seq: int) -> None:
        """
        시퀀스 번호 기반 패킷 손실 추정.
        
        시퀀스 번호는 0-255 순환이므로 롤오버를 고려합니다.
        
        Args:
            current_seq: 현재 패킷의 시퀀스 번호
        """
        if self._last_seq is not None:
            # 예상 시퀀스 (이전 + 1, 256에서 롤오버)
            expected = (self._last_seq + 1) % 256
            
            if current_seq != expected:
                # 손실 수 계산 (롤오버 고려)
                if current_seq > expected:
                    gap = current_seq - expected
                else:
                    gap = (256 - expected) + current_seq
                
                # 큰 갭은 재시작으로 간주 (손실로 카운트하지 않음)
                if gap < 100:
                    self._packet_loss += gap
                    logger.debug(
                        f"Packet loss detected: expected={expected}, "
                        f"got={current_seq}, gap={gap}"
                    )
        
        self._last_seq = current_seq
    
    def _update_jitter(self, now: datetime) -> None:
        """
        지터 계산.
        
        연속된 패킷 간의 도착 시간 차이를 기반으로 지터를 계산합니다.
        
        Args:
            now: 현재 패킷 수신 시간
        """
        if len(self._packet_times) > 0:
            # 이전 패킷과의 시간 차이 (밀리초)
            prev_time = self._packet_times[-1]
            delta_ms = (now - prev_time).total_seconds() * 1000
            
            # 예상 간격 (1000ms / expected_pps)
            expected_interval_ms = 1.0  # 대략 1ms 간격 가정
            
            # 지터 = 실제 간격 - 예상 간격의 절대값
            jitter = abs(delta_ms - expected_interval_ms)
            self._jitter_values.append(jitter)
        
        self._packet_times.append(now)
    
    def _update_state_from_payload(self, payload) -> None:
        """
        상태보고 페이로드에서 상태 업데이트.
        
        Args:
            payload: StatusReportPayload 인스턴스
        """
        self._latest_state = {
            "operationalMode": payload.operation_mode,
            "operationalAuthority": payload.authority,
            "drivingState": payload.driving_state,
            "devices": payload.device_presence,
            "emergencyStatus": payload.emergency_causes,
        }
    
    def _add_log_entry(self, result: ParseResult, now: datetime) -> None:
        """
        로그 엔트리 추가.
        
        Args:
            result: 파싱 결과
            now: 수신 시간
        """
        entry = LogEntry(
            timestamp=now,
            sequence=result.header.sequence if result.header else 0,
            msg_code=f"0x{result.header.msg_code:02X}" if result.header else "---",
            parse_ok=result.success,
            checksum_ok=result.checksum_ok,
            mode=self._latest_state.get("operationalMode", "---"),
            authority=self._latest_state.get("operationalAuthority", "---"),
            notes=result.error if not result.success else "",
        )
        self._logs.appendleft(entry)
    
    def get_dashboard_data(self) -> Dict:
        """
        대시보드 데이터 반환.
        
        MockDataGenerator.generate_initial_data()와 동일한 형식으로
        현재 상태를 반환합니다.
        
        Returns:
            대시보드 상태 딕셔너리
        """
        # 먼저 대기 중인 패킷 처리
        self.process_pending_packets()
        
        # 차트 데이터 포인트 추가
        self._update_chart_data()
        
        # 가용성 업데이트
        self._update_availability()
        
        with self._lock:
            # 캡처 통계
            capture_stats = {}
            if self._sniffer:
                stats = self._sniffer.get_stats()
                capture_stats = stats.to_dict()
            
            # 지터 P95/P99 계산
            jitter_p95 = self._calculate_percentile(95)
            jitter_p99 = self._calculate_percentile(99)
            
            return {
                "connected": self._connected,
                "interface": self._interface,
                "filter": f"{self._src_port}→{self._dst_port}",
                "lastPacketTime": (
                    self._last_packet_time.strftime("%H:%M:%S")
                    if self._last_packet_time else ""
                ),
                "capturePps": capture_stats.get("pps", 0),
                "filterPass": capture_stats.get("filter_pass_pct", 100.0),
                "parseSuccess": self._get_parse_success_pct(),
                "checksumFail": self._get_checksum_fail_pct(),
                "packetLoss": self._packet_loss,
                "availability5min": self._calculate_availability(300),
                "availability1hour": self._calculate_availability(3600),
                "jitterP95": round(jitter_p95, 1),
                "jitterP99": round(jitter_p99, 1),
                **self._latest_state,  # 운용상태, 장치상태 등
                "combinedData": list(self._chart_data),
                "availabilitySegments": [
                    s.to_dict() for s in self._availability_segments
                ],
            }
    
    def _update_chart_data(self) -> None:
        """차트 데이터 포인트 추가."""
        now = datetime.now()
        
        capture_stats = {}
        if self._sniffer:
            capture_stats = self._sniffer.get_stats().to_dict()
        
        jitter_values = list(self._jitter_values)
        avg_jitter = sum(jitter_values) / len(jitter_values) if jitter_values else 0
        
        self._chart_data.append({
            "timestamp": now.strftime("%H:%M:%S"),
            "pps": capture_stats.get("pps", 0),
            "jitter": round(avg_jitter, 2),
        })
    
    def _update_availability(self) -> None:
        """가용성 타임라인 업데이트."""
        self._elapsed_time += config.ui.poll_interval_ms / 1000
        time_window = config.ui.timeline_duration_sec
        
        is_connected = self._connected
        
        if not self._availability_segments:
            self._availability_segments.append(
                AvailabilitySegment(0, time_window, is_connected)
            )
            return
        
        last_seg = self._availability_segments[-1]
        
        # 상태 변경 여부에 따라 세그먼트 확장 또는 생성
        if (is_connected and last_seg.is_up) or (not is_connected and not last_seg.is_up):
            last_seg.end_sec = time_window
        else:
            self._availability_segments.append(
                AvailabilitySegment(
                    start_sec=last_seg.end_sec,
                    end_sec=time_window,
                    is_up=is_connected,
                )
            )
        
        # 오래된 세그먼트 정리
        if self._elapsed_time > time_window:
            offset = self._elapsed_time - time_window
            pruned = []
            for seg in self._availability_segments:
                new_start = max(0, seg.start_sec - offset)
                new_end = max(0, seg.end_sec - offset)
                if new_end > 0:
                    pruned.append(
                        AvailabilitySegment(
                            start_sec=new_start,
                            end_sec=new_end,
                            is_up=seg.is_up,
                        )
                    )
            if pruned:
                pruned[-1].end_sec = time_window
            self._availability_segments = pruned
    
    def _calculate_availability(self, window_sec: int) -> float:
        """
        가용성 백분율 계산.
        
        Args:
            window_sec: 시간 윈도우 (초)
            
        Returns:
            가용성 백분율 (0-100)
        """
        if not self._availability_segments:
            return 100.0
        
        up_time = 0
        for seg in self._availability_segments:
            if seg.is_up:
                # 윈도우 내 up 시간 합산
                effective_start = max(0, seg.start_sec)
                effective_end = min(window_sec, seg.end_sec)
                if effective_end > effective_start:
                    up_time += effective_end - effective_start
        
        total_time = min(self._elapsed_time, window_sec)
        if total_time <= 0:
            return 100.0
        
        return round((up_time / total_time) * 100, 2)
    
    def _calculate_percentile(self, percentile: int) -> float:
        """
        지터 백분위수 계산.
        
        Args:
            percentile: 백분위수 (예: 95, 99)
            
        Returns:
            지터 값 (밀리초)
        """
        values = sorted(self._jitter_values)
        if not values:
            return 0.0
        
        idx = int(len(values) * percentile / 100)
        idx = min(idx, len(values) - 1)
        return values[idx]
    
    def _get_parse_success_pct(self) -> float:
        """파싱 성공률 계산."""
        if self._parse_total == 0:
            return 100.0
        return round((self._parse_success / self._parse_total) * 100, 1)
    
    def _get_checksum_fail_pct(self) -> float:
        """체크섬 실패율 계산."""
        if self._parse_total == 0:
            return 0.0
        return round((self._checksum_fail / self._parse_total) * 100, 1)
    
    def get_logs(self, limit: int = 50) -> List[Dict]:
        """
        로그 엔트리 반환.
        
        Args:
            limit: 최대 반환 개수
            
        Returns:
            로그 딕셔너리 리스트
        """
        with self._lock:
            logs = list(self._logs)[:limit]
            return [log.to_dict() for log in logs]
    
    def clear_logs(self) -> None:
        """로그 초기화."""
        with self._lock:
            self._logs.clear()
    
    def reset_stats(self) -> None:
        """통계 초기화."""
        with self._lock:
            self._parse_total = 0
            self._parse_success = 0
            self._checksum_fail = 0
            self._packet_loss = 0
            self._last_seq = None
            self._jitter_values.clear()
            self._packet_times.clear()


# 전역 인스턴스 (싱글톤 패턴)
_live_provider: Optional[LiveDataProvider] = None


def get_live_provider() -> LiveDataProvider:
    """
    LiveDataProvider 싱글톤 인스턴스 반환.
    
    Returns:
        LiveDataProvider 인스턴스
    """
    global _live_provider
    if _live_provider is None:
        _live_provider = LiveDataProvider()
    return _live_provider
```

```

│ PacketSniffer   │────▶│ PacketQueue  │────▶│ ICDParser  │

                        └──────────────────────────────────┘
                                                   │
                                                   ▼
                        ┌──────────────────────────────────┐
                        │      get_dashboard_data()        │
                        │   (MockDataGenerator와 동일한     │
                        │    인터페이스 제공)               │
                        └──────────────────────────────────┘
```
