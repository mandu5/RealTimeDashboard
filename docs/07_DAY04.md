### 1. `ugv_mon/analysis/quality.py` - 품질 분석기

**역할**: 파싱 결과를 분석하여 데이터 품질 지표 산출

**구현 내용**:

```python
"""
데이터 품질 분석 모듈

역할:
- 파싱 성공률 추적
- 체크섬 실패율 추적
- 길이 불일치 감지
- Unknown 메시지 타입 카운트
- 구조화된 로그 엔트리 생성
"""

from dataclasses import dataclass
from collections import deque
from typing import List, Dict

from ..parser.icd_parser import ParseResult


@dataclass
class QualityMetrics:
    """
    품질 메트릭 스냅샷
    
    Attributes:
        total_packets: 총 처리된 패킷 수
        parse_success: 파싱 성공 횟수
        parse_fail: 파싱 실패 횟수
        checksum_ok: 체크섬 검증 성공 횟수
        checksum_fail: 체크섬 검증 실패 횟수
        length_mismatch: 길이 불일치 횟수
        unknown_type: 알 수 없는 메시지 타입 횟수
    """
    total_packets: int = 0
    parse_success: int = 0
    parse_fail: int = 0
    checksum_ok: int = 0
    checksum_fail: int = 0
    length_mismatch: int = 0
    unknown_type: int = 0
    
    @property
    def parse_success_rate(self) -> float:
        """파싱 성공률 (%)"""
        if self.total_packets == 0:
            return 100.0
        return (self.parse_success / self.total_packets) * 100.0
    
    @property
    def checksum_fail_rate(self) -> float:
        """체크섬 실패율 (%)"""
        if self.total_packets == 0:
            return 0.0
        return (self.checksum_fail / self.total_packets) * 100.0
    
    @property
    def unknown_rate(self) -> float:
        """Unknown 메시지 타입 비율 (%)"""
        if self.total_packets == 0:
            return 0.0
        return (self.unknown_type / self.total_packets) * 100.0


class QualityAnalyzer:
    """
    데이터 품질 분석기
    
    파싱 결과를 기록하고 품질 메트릭을 계산합니다.
    롤링 윈도우를 사용하여 최근 N개 패킷에 대한 메트릭을 추적합니다.
    
    사용법:
        analyzer = QualityAnalyzer(window_size=1000)
        result = parser.parse(packet_data)
        analyzer.record(result)
        metrics = analyzer.get_metrics()
    """
    
    def __init__(self, window_size: int = 1000):
        """
        초기화
        
        Args:
            window_size: 롤링 윈도우 크기 (최근 N개 패킷만 추적)
        """
        self.window_size = window_size
        self._results: deque = deque(maxlen=window_size)
        self._metrics = QualityMetrics()
        
    def record(self, result: ParseResult) -> None:
        """
        파싱 결과 기록
        
        Args:
            result: ParseResult 객체
        """
        self._results.append(result)
        self._update_metrics(result)
    
    def _update_metrics(self, result: ParseResult) -> None:
        """
        메트릭 업데이트
        
        Args:
            result: ParseResult 객체
        """
        self._metrics.total_packets += 1
        
        # 파싱 성공/실패
        if result.success:
            self._metrics.parse_success += 1
        else:
            self._metrics.parse_fail += 1
            
            # 에러 타입별 분류
            if "Length mismatch" in result.error:
                self._metrics.length_mismatch += 1
        
        # 체크섬 검증
        if result.checksum_ok:
            self._metrics.checksum_ok += 1
        else:
            self._metrics.checksum_fail += 1
        
        # Unknown 타입 체크 (헤더는 파싱되었지만 알 수 없는 메시지 코드)
        if result.header and result.header.msg_code not in [0x01]:  # 0x01 = STATUS_REPORT
            self._metrics.unknown_type += 1
    
    def get_metrics(self) -> QualityMetrics:
        """
        현재 메트릭 반환
        
        Returns:
            QualityMetrics 객체
        """
        return self._metrics
    
    def get_metrics_dict(self) -> Dict:
        """
        대시보드용 메트릭 딕셔너리 반환
        
        Returns:
            메트릭 딕셔너리 (KPI 카드용)
        """
        m = self._metrics
        return {
            "parseSuccess": round(m.parse_success_rate, 1),
            "checksumFail": round(m.checksum_fail_rate, 1),
            "totalPackets": m.total_packets,
            "unknownCount": m.unknown_type,
            "lengthMismatchCount": m.length_mismatch,
        }
    
    def reset(self) -> None:
        """메트릭 리셋 (테스트용)"""
        self._results.clear()
        self._metrics = QualityMetrics()
```

**설명 및 근거**:
- **롤링 윈도우**: 최근 N개 패킷만 추적하여 메모리 사용량 제한
- **Property 기반 계산**: 메트릭은 필요할 때 계산 (성능 최적화)
- **구조화된 메트릭**: dataclass로 타입 안전성 보장
- **에러 분류**: 에러 타입별 카운팅으로 문제 진단 용이

## 🧪 테스트 방법

### 1. 단위 테스트

```python
# test_quality.py
from ugv_mon.analysis.quality import QualityAnalyzer, ParseResult, ICDHeader

def test_quality_tracking():
    """품질 메트릭 추적 테스트"""
    analyzer = QualityAnalyzer(window_size=100)
    
    # 성공 케이스
    success_result = ParseResult(
        success=True,
        header=ICDHeader(...),
        checksum_ok=True,
    )
    analyzer.record(success_result)
    
    # 실패 케이스 (체크섬 오류)
    fail_result = ParseResult(
        success=False,
        checksum_ok=False,
        error="Checksum mismatch",
    )
    analyzer.record(fail_result)
    
    metrics = analyzer.get_metrics()
    assert metrics.total_packets == 2
    assert metrics.parse_success == 1
    assert metrics.parse_fail == 1
    assert metrics.checksum_fail == 1
    assert metrics.parse_success_rate == 50.0
```

### 2. 통합 테스트

```bash
# 캡처 + 파싱 + 품질 분석 테스트
sudo python -c "
from ugv_mon.capture.sniffer import PacketSniffer
from ugv_mon.parser.icd_parser import ICDParser
from ugv_mon.analysis.quality import QualityAnalyzer

sniffer = PacketSniffer('lo')
parser = ICDParser()
analyzer = QualityAnalyzer()

sniffer.start()

import time
start_time = time.time()
while time.time() - start_time < 10:  # 10초간
    packet = sniffer.get_packet(timeout=0.5)
    if packet:
        payload = bytes(packet['UDP'].payload)
        result = parser.parse(payload)
        analyzer.record(result)

metrics = analyzer.get_metrics()
print(f'Total packets: {metrics.total_packets}')
print(f'Parse success rate: {metrics.parse_success_rate:.1f}%')
print(f'Checksum fail rate: {metrics.checksum_fail_rate:.1f}%')
print(f'Unknown types: {metrics.unknown_type}')

sniffer.stop()
"
```
