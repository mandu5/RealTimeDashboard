### 1. `ugv_mon/data/models.py` - Time 포맷 변경

```python
@dataclass
class LogEntry:
    """로그 엔트리."""
    timestamp: datetime
    sequence: int
    msg_code: str
    parse_ok: bool = True
    checksum_ok: bool = True
    mode: str = ""
    authority: str = ""
    notes: str = ""
    
    def to_dict(self) -> Dict:
        """딕셔너리 변환."""
        return {
            "time": self.timestamp.strftime("%H:%M:%S"),  # 밀리초 제거
            "seq": self.sequence,
            "msg_code": self.msg_code,
            "parse_ok": "✓" if self.parse_ok else "✗",
            "checksum_ok": "✓" if self.checksum_ok else "✗",
            "mode": self.mode,
            "authority": self.authority,
            "notes": self.notes,
        }
```

### 2. `ugv_mon/components/device_grid.py` - Tooltip 추가

```python
"""
Device Grid Component.

장치 연결 상태 그리드를 생성합니다.

특징:
- 연결 상태에 따른 색상 (초록/노랑/빨강)
- 경고/오류 시 hover tooltip 표시
- 모든 박스 크기 동일 유지
"""

from dash import html
from typing import Dict, List
import dash_mantine_components as dmc


def create_device_grid(devices: List[Dict]) -> html.Div:
    """
    장치 연결 상태 그리드 생성.
    
    Args:
        devices: 장치 상태 리스트
                 [{"id": "vic", "name": "VIC", "connected": True, 
                   "warning": False, "error_reason": None}, ...]
    
    Returns:
        Dash HTML Div 컴포넌트
    """
    items = [create_device_item(d) for d in devices]
    
    return html.Div(
        id="device-grid",
        children=items,
        style={
            "display": "grid",
            "gridTemplateColumns": "repeat(5, 1fr)",
            "gap": "8px",
        }
    )


def create_device_item(device: Dict) -> html.Div:
    """
    개별 장치 상태 아이템 생성.
    
    Args:
        device: 장치 상태 딕셔너리
    
    Returns:
        Dash HTML Div 컴포넌트
    """
    name = device.get("name", "???")
    connected = device.get("connected", False)
    warning = device.get("warning", False)
    error_reason = device.get("error_reason")
    
    # 상태에 따른 아이콘 및 스타일
    if not connected:
        icon = "🔴"
        style = {
            "backgroundColor": "#fef2f2",
            "borderColor": "#fecaca",
            "color": "#dc2626",
        }
    elif warning:
        icon = "🟡"
        style = {
            "backgroundColor": "#fefce8",
            "borderColor": "#fef08a",
            "color": "#ca8a04",
        }
    else:
        icon = "🟢"
        style = {
            "backgroundColor": "#f0fdf4",
            "borderColor": "#bbf7d0",
            "color": "#16a34a",
        }
    
    # 기본 박스 스타일 (모든 박스 동일 크기)
    base_style = {
        "padding": "8px 12px",
        "borderRadius": "8px",
        "textAlign": "center",
        "width": "100%",
        "minHeight": "40px",
        "boxSizing": "border-box",
        "display": "flex",
        "alignItems": "center",
        "justifyContent": "center",
        "border": "1px solid",
        **style,
    }
    
    # HTML title 속성으로 tooltip (레이아웃 영향 없음)
    props = {"style": base_style}
    if error_reason and (not connected or warning):
        props["title"] = error_reason
    
    return html.Div(
        children=[
            html.Span(
                icon,
                style={"fontSize": "14px", "marginRight": "6px"}
            ),
            html.Span(
                name,
                style={"fontSize": "12px", "fontWeight": "600"}
            ),
        ],
        **props,
    )
```

"""

def create_log_table(initial_data: List[Dict]) -> dag.AgGrid:
