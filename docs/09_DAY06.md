### 1. `ugv_mon/data/live_provider.py` 수정

> 기존 코드에 분석 모듈을 통합합니다.

**추가/수정할 부분**:

```python
# 기존 import에 추가
from ..analysis import StatsCalculator, AnomalyDetector

class LiveDataProvider:
    """수정된 LiveDataProvider."""
    
    def __init__(self, ...):
        # 기존 초기화 코드...
        
        # 분석 모듈 추가
        self._stats_calc = StatsCalculator(window_sec=60)
        self._anomaly_detector = AnomalyDetector()
        self._recent_anomalies: deque = deque(maxlen=50)
    
    def _process_single_packet(self, raw_data: bytes) -> None:
        """수정된 패킷 처리."""
        now = datetime.now()
        
        with self._lock:
            # 기존 파싱 코드...
            result = self._parser.parse(raw_data)
            
            if result.success and result.header:
                # StatsCalculator에 기록
                jitter = self._stats_calc.record_packet(
                    timestamp=now,
                    sequence=result.header.sequence,
                    size=len(raw_data)
                )
                
                # 이상 탐지 (지터 기반)
                if jitter is not None and jitter > self._anomaly_detector._jitter_warning:
                    self._add_anomaly_log(f"높은 지터 감지: {jitter:.1f}ms")
            
            # 기존 나머지 코드...
    
    def get_dashboard_data(self) -> Dict:
        """수정된 대시보드 데이터."""
        self.process_pending_packets()
        
        with self._lock:
            # StatsCalculator에서 통계 가져오기
            p95, p99 = self._stats_calc.get_jitter_percentiles()
            pps = self._stats_calc.get_pps()
            loss = self._stats_calc.get_packet_loss()
            avail_5min = self._stats_calc.get_availability(300)
            avail_1hour = self._stats_calc.get_availability(3600)
            
            # 이상 탐지
            anomalies = self._anomaly_detector.check_all(
                jitter_p95=p95,
                jitter_p99=p99,
                loss_count=loss,
                last_packet_time=self._last_packet_time,
                availability_pct=avail_5min,
                checksum_fail_pct=self._get_checksum_fail_pct(),
            )
            
            # 새 이상 로그에 추가
            for anomaly in anomalies:
                self._add_anomaly_log(anomaly.message)
            
            return {
                # 기존 필드들...
                "jitterP95": round(p95, 1),
                "jitterP99": round(p99, 1),
                "capturePps": pps,
                "packetLoss": loss,
                "availability5min": avail_5min,
                "availability1hour": avail_1hour,
                "anomalyCount": len(anomalies),
                # ...
            }
    
    def _add_anomaly_log(self, message: str) -> None:
        """이상 탐지 로그 추가."""
        now = datetime.now()
        entry = LogEntry(
            timestamp=now,
            sequence=0,
            msg_code="---",
            parse_ok=True,
            checksum_ok=True,
            mode="---",
            authority="---",
            notes=f"⚠️ {message}",
        )
        self._logs.appendleft(entry)
```

### 2. `ugv_mon/callbacks/update_callbacks.py` 개선

> 환경변수로 데이터 소스를 전환할 수 있도록 개선합니다.

```python
"""
Update Callbacks Module.

대시보드의 모든 상호작용을 처리합니다.
데이터 소스는 환경변수로 전환 가능합니다.

환경변수:
- UGV_MON_USE_LIVE=true: 실시간 캡처 모드
- UGV_MON_USE_LIVE=false (기본값): Mock 데이터 모드
"""

import os
from dash import callback_context
from dash.dependencies import Input, Output, State
from dash.exceptions import PreventUpdate
from typing import Dict, List, Tuple

# 환경변수로 데이터 소스 선택
USE_LIVE_DATA = os.getenv("UGV_MON_USE_LIVE", "false").lower() == "true"

if USE_LIVE_DATA:
    from ..data.live_provider import LiveDataProvider
    
    _data_provider = None
    
    def get_data_generator():
        """LiveDataProvider 싱글톤 반환."""
        global _data_provider
        if _data_provider is None:
            _data_provider = LiveDataProvider()
            _data_provider.start_capture()
        return _data_provider
else:
    from ..data.mock_data import MockDataGenerator
    
    _data_provider = None
    
    def get_data_generator():
        """MockDataGenerator 싱글톤 반환."""
        global _data_provider
        if _data_provider is None:
            _data_provider = MockDataGenerator()
        return _data_provider


def register_callbacks(app) -> None:
    """
    모든 콜백 등록.
    
    Args:
        app: Dash 앱 인스턴스
    """
    _register_polling_callback(app)
    _register_update_callback(app)
    _register_control_callbacks(app)
    _register_chart_time_range_callback(app)


def _register_polling_callback(app) -> None:
    """데이터 폴링 콜백."""
    
    @app.callback(
        Output("dashboard-data", "data"),
        [Input("poll-interval", "n_intervals")],
        [State("is-paused", "data"), State("dashboard-data", "data")],
    )
    def update_dashboard_data(
        n_intervals: int,
        is_paused: bool,
        prev_data: Dict
    ) -> Dict:
        """
        2초마다 대시보드 데이터 갱신.
        
        일시정지 상태면 이전 데이터 유지.
        """
        if is_paused:
            raise PreventUpdate
        
        gen = get_data_generator()
        
        if USE_LIVE_DATA:
            # Live 모드: get_dashboard_data() 호출
            return gen.get_dashboard_data()
        else:
            # Mock 모드: 기존 방식
            if prev_data is None:
                return gen.generate_initial_data()
            return gen.update_data(prev_data)


def _register_update_callback(app) -> None:
    """UI 업데이트 콜백."""
    
    @app.callback(
        [
            # KPI 카드들
            Output("kpi-capture-pps", "children"),
            Output("kpi-filter-pass", "children"),
            Output("kpi-parse-success", "children"),
            Output("kpi-checksum-fail", "children"),
            Output("kpi-packet-loss", "children"),
            Output("kpi-avail-5min", "children"),
            Output("kpi-avail-1hour", "children"),
            Output("kpi-jitter-p95", "children"),
            # 상태 칩
            Output("status-connected", "children"),
            Output("status-last-packet", "children"),
            # 운용상태
            Output("op-mode", "children"),
            Output("op-authority", "children"),
            Output("op-driving", "children"),
            # 장치 그리드
            Output("device-grid", "children"),
            # 비상정지
            Output("emergency-grid", "children"),
            # 차트
            Output("comm-quality-chart", "figure"),
            Output("availability-chart", "figure"),
            # 로그 테이블
            Output("log-table", "rowData"),
            # 실시간 값 표시
            Output("current-pps-display", "children"),
            Output("current-jitter-display", "children"),
        ],
        [
            Input("dashboard-data", "data"),
            Input("chart-time-range", "data"),
        ],
    )
    def update_all_components(
        data: Dict,
        time_range: int
    ) -> Tuple:
        """데이터 변경 시 모든 UI 컴포넌트 갱신."""
        if data is None:
            raise PreventUpdate
        
        # 기존 업데이트 로직...
        # (이전 코드와 동일)
        pass


def _register_control_callbacks(app) -> None:
    """컨트롤 버튼 콜백."""
    
    @app.callback(
        Output("is-paused", "data"),
        [Input("pause-btn", "n_clicks")],
        [State("is-paused", "data")],
        prevent_initial_call=True,
    )
    def toggle_pause(n_clicks: int, is_paused: bool) -> bool:
        """일시정지 토글."""
        return not is_paused
    
    @app.callback(
        Output("log-table", "rowData", allow_duplicate=True),
        [Input("clear-btn", "n_clicks")],
        prevent_initial_call=True,
    )
    def clear_logs(n_clicks: int) -> List:
        """로그 초기화."""
        gen = get_data_generator()
        gen.clear_logs()
        return []


def _register_chart_time_range_callback(app) -> None:
    """차트 시간 범위 선택 콜백."""
    
    @app.callback(
        Output("chart-time-range", "data"),
        [
            Input("time-range-30s-btn", "n_clicks"),
            Input("time-range-1m-btn", "n_clicks"),
            Input("time-range-5m-btn", "n_clicks"),
        ],
        prevent_initial_call=True,
    )
    def update_time_range(n_30s, n_1m, n_5m) -> int:
        """시간 범위 버튼 클릭 처리."""
        ctx = callback_context
        if not ctx.triggered:
            raise PreventUpdate
        
        button_id = ctx.triggered[0]["prop_id"].split(".")[0]
        
        if button_id == "time-range-30s-btn":
            return 30
        elif button_id == "time-range-1m-btn":
            return 60
        elif button_id == "time-range-5m-btn":
            return 300
        
        return 60  # 기본값
```

```bash
sudo python -c "
from scapy.all import *
import struct

# ICD 형식의 테스트 패킷 생성
header = struct.pack('<IBBBBHH', 
    0x12345678,  # timestamp
    0x01, 0x02,  # src, dst
    0x01, 0x00,  # msg_code, ack_code
    0x0000, 87   # reserved, data_length
)
payload = b'\\x45' + b'\\x00' * 5 + b'\\xFF\\x03' + b'\\x00' * 12 + b'\\x00\\x00' + b'\\x00' * 65
data = header + payload
checksum = struct.pack('<H', sum(data) & 0xFFFF)
packet = data + checksum

pkt = IP(dst='127.0.0.1')/UDP(sport=50000, dport=61000)/Raw(load=packet)
send(pkt, verbose=False, count=100, inter=0.01)
print('Sent 100 ICD packets')
"
```
