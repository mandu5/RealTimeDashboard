### 3. `tests/test_seq_rollover.py` - 시퀀스 롤오버 테스트

```python
"""
시퀀스 롤오버 테스트
"""

import pytest
from ugv_mon.analysis.loss import LossEstimator


def test_rollover_normal():
    """정상 롤오버 (255 -> 0)"""
    estimator = LossEstimator()
    
    estimator.record_sequence(255)
    loss = estimator.record_sequence(0)
    
    assert loss == 1  # 정상 (롤오버)


def test_rollover_with_loss():
    """롤오버 + 손실 (254 -> 2)"""
    estimator = LossEstimator()
    
    estimator.record_sequence(254)
    loss = estimator.record_sequence(2)
    
    assert loss == 4  # 3개 손실 (255, 0, 1)
```
